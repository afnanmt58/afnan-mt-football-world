const PlayerAPI = {
  provider: 'TheSportsDB',
  baseUrl: 'https://www.thesportsdb.com/api/v1/json/123',
  async search(query){
    const url = `${this.baseUrl}/searchplayers.php?p=${encodeURIComponent(query)}`;
    const res = await fetch(url);
    if(!res.ok) throw new Error('API error: '+res.status);
    const data = await res.json();
    return data.player || [];
  },
  async lookup(id){
    const url = `${this.baseUrl}/lookupplayer.php?id=${encodeURIComponent(id)}`;
    const res = await fetch(url);
    if(!res.ok) throw new Error('API error: '+res.status);
    const data = await res.json();
    return (data.players && data.players[0]) || null;
  }
};

/* ==================================================================
   SEARCH + RENDER
   ================================================================== */
async function performPlayerSearch(){
  const query = document.getElementById('playerSearchInput').value.trim();
  if(!query){
    showSearchStatus('Please enter a player name.', 'empty');
    return;
  }
  showSearchStatus('<i class="fas fa-spinner"></i> Searching TheSportsDB...', 'loading');
  document.getElementById('playerGrid').innerHTML = '';
  try{
    const results = await PlayerAPI.search(query);
    apiAvailable = true;
    if(!results || results.length === 0){
      currentSearchResults = null;
      showSearchStatus('No players found for "'+escapeHtml(query)+'". Try a different spelling or check the name.', 'empty');
      renderPlayerGrid(localGoats);
      return;
    }
    currentSearchResults = results;
    showSearchStatus(`Found ${results.length} player${results.length>1?'s':''} for "${escapeHtml(query)}"`, '');
    renderPlayerGrid(results.map(r=>({
      name: r.strPlayer,
      nation: r.strNationality || 'Information unavailable',
      club: r.strTeam || 'Information unavailable',
      position: r.strPosition || 'Information unavailable',
      img: r.strThumb || r.strCutout || '',
      id: r.idPlayer,
      isApi: true
    })));
    populateFilterOptions(results);
  }catch(err){
    console.warn('API error:', err);
    apiAvailable = false;
    currentSearchResults = null;
    showSearchStatus('⚠️ API unavailable right now. Showing local legends instead.', 'error');
    renderPlayerGrid(localGoats);
  }
}

function renderPlayerGrid(list){
  const grid = document.getElementById('playerGrid');
  if(!grid) return;
  if(!list || list.length === 0){
    grid.innerHTML = '<div class="search-status empty">No players to display.</div>';
    return;
  }
  currentDisplayList = list;
  grid.innerHTML = list.map(p=>{
    const img = p.img ? `<img src="${p.img}" alt="${escapeHtml(p.name)}" loading="lazy" onerror="this.style.display='none';this.nextElementSibling.style.display='flex'"><div class="pc-placeholder" style="display:none"><i class="fas fa-user"></i></div>` : `<div class="pc-placeholder"><i class="fas fa-user"></i></div>`;
    return `
      <div class="player-card" onclick="openPlayerProfile(${p.isApi ? `'${p.id}',true` : `null,false,${p.rank||0}`})">
        <div class="pc-img-wrap">${img}${p.rank ? `<span class="pc-goat-rank">#${p.rank}</span>`:''}</div>
        <div class="pc-info">
          <div class="pc-name">${escapeHtml(p.name)}</div>
          <div class="pc-meta"><strong>${escapeHtml(p.position)}</strong></div>
          <div class="pc-meta">${escapeHtml(p.club)}</div>
          <span class="pc-badge">${escapeHtml(p.nation)}</span>
        </div>
      </div>
    `;
  }).join('');
}

function showSearchStatus(html, cls){
  const el = document.getElementById('searchStatus');
  if(!el) return;
  el.className = 'search-status ' + (cls||'');
  el.innerHTML = html;
}

function populateFilterOptions(results){
  const nats = [...new Set(results.map(r=>r.strNationality).filter(Boolean))].sort();
  const clubs = [...new Set(results.map(r=>r.strTeam).filter(Boolean))].sort();
  const positions = [...new Set(results.map(r=>r.strPosition).filter(Boolean))].sort();
  const leagues = [...new Set(results.map(r=>r.strLeague).filter(Boolean))].sort();
  const fill = (id, arr) => {
    const el = document.getElementById(id);
    if(!el) return;
    el.innerHTML = '<option value="">All</option>' + arr.map(v=>`<option value="${escapeHtml(v)}">${escapeHtml(v)}</option>`).join('');
  };
  fill('filterNationality', nats);
  fill('filterClub', clubs);
  fill('filterPosition', positions);
  fill('filterLeague', leagues);
}

function applyLocalFilters(){
  if(!currentSearchResults) return;
  const nat = document.getElementById('filterNationality').value;
  const club = document.getElementById('filterClub').value;
  const pos = document.getElementById('filterPosition').value;
  const league = document.getElementById('filterLeague').value;
  let filtered = currentSearchResults.filter(r=>{
    if(nat && r.strNationality !== nat) return false;
    if(club && r.strTeam !== club) return false;
    if(pos && r.strPosition !== pos) return false;
    if(league && r.strLeague !== league) return false;
    return true;
  });
  if(filtered.length === 0){
    showSearchStatus('No players match your filters.', 'empty');
    renderPlayerGrid([]);
    return;
  }
  showSearchStatus(`Showing ${filtered.length} filtered result${filtered.length>1?'s':''}`, '');
  renderPlayerGrid(filtered.map(r=>({
    name: r.strPlayer,
    nation: r.strNationality || 'Information unavailable',
    club: r.strTeam || 'Information unavailable',
    position: r.strPosition || 'Information unavailable',
    img: r.strThumb || r.strCutout || '',
    id: r.idPlayer,
    isApi: true
  })));
}

async function openPlayerProfile(id, isApi, localRank){
  const modal = document.getElementById('playerProfileModal');
  const content = document.getElementById('playerProfileContent');
  content.innerHTML = `<button class="player-profile-close" onclick="closePlayerProfile()">&times;</button><div style="padding:40px;text-align:center;"><i class="fas fa-spinner fa-spin" style="font-size:2rem;color:var(--gold);"></i><p style="margin-top:12px;color:var(--text-muted);font-weight:700;">Loading player profile...</p></div>`;
  modal.classList.add('active');
  document.body.style.overflow='hidden';
  try{
    let p;
    if(isApi){
      p = await PlayerAPI.lookup(id);
      if(!p) throw new Error('Player not found');
      renderApiProfile(p);
    } else {
      const local = localGoats.find(g=>g.rank===localRank);
      if(!local) throw new Error('Local player not found');
      renderLocalProfile(local);
    }
  }catch(err){
    console.warn('Profile load error:', err);
    content.innerHTML = `<button class="player-profile-close" onclick="closePlayerProfile()">&times;</button><div style="padding:40px;text-align:center;"><i class="fas fa-exclamation-triangle" style="font-size:2.5rem;color:#e05555;"></i><h3 style="color:var(--bright-gold);margin:12px 0;">Profile Unavailable</h3><p style="color:var(--text-muted);font-weight:600;">${escapeHtml(err.message)}</p><p style="color:var(--text-muted);font-size:0.85rem;margin-top:8px;">Player details could not be retrieved from the API.</p><button class="btn-premium" style="margin-top:16px;" onclick="closePlayerProfile()">Close</button></div>`;
  }
}

  const color=colors[Math.floor(Math.random()*colors.length)];
  const x=Math.random()*(rect.width-55), y=Math.random()*(rect.height-55-50)+25;
  const t=document.createElement('div'); t.className='shooter-target';
  t.style.cssText=`left:${x}px;top:${y}px;background:radial-gradient(circle,${color},${color}88);box-shadow:0 0 30px ${color}88;border:3px solid ${color};`;
  t.textContent=emojis[Math.floor(Math.random()*emojis.length)];
  area.appendChild(t);
  setTimeout(()=>{
    if(t.parentElement&&!t.classList.contains('hit')){
      t.style.transition='all 0.3s'; t.style.transform='scale(0)'; t.style.opacity='0';
      setTimeout(()=>t.remove(),300); sCombo=0; const c=document.getElementById('sCombo'); if(c) c.textContent='0';
    }
  },2500);
}

  memCards=emojis.map((e,i)=>({id:i,emoji:e,flipped:false,matched:false}));
  memFlip=[]; memMatched=0; memLocked=false; memMoves=0;
  for(let i=memCards.length-1;i>0;i--){ const j=Math.floor(Math.random()*(i+1)); [memCards[i],memCards[j]]=[memCards[j],memCards[i]]; }
  area.innerHTML=`<div class="game-stats"><span class="stat">🧠 Moves: <span class="val" id="memMoves">0</span></span><span class="stat">✅ Matched: <span class="val" id="memMatched">0/8</span></span></div><div class="memory-grid" id="memoryGrid"></div>`;
  const grid=document.getElementById('memoryGrid');
  memCards.forEach((c,i)=>{ const cell=document.createElement('div'); cell.className='memory-cell hidden'; cell.dataset.index=i; cell.textContent='❓'; cell.addEventListener('click',()=>flipMem(i)); grid.appendChild(cell); });
}
function flipMem(index){
  if(memLocked) return;
  const card=memCards[index]; if(card.flipped||card.matched) return; if(memFlip.length>=2) return;
  card.flipped=true;
  const cells=document.querySelectorAll('.memory-cell');
  cells[index].textContent=card.emoji; cells[index].classList.remove('hidden'); cells[index].classList.add('flipped');
  memFlip.push(index);
  if(memFlip.length===2){
    memMoves++; document.getElementById('memMoves').textContent=memMoves;
    memLocked=true; const [i1,i2]=memFlip;
    if(memCards[i1].emoji===memCards[i2].emoji){
      memCards[i1].matched=true; memCards[i2].matched=true;
      cells[i1].classList.add('matched'); cells[i2].classList.add('matched');
      memFlip=[]; memLocked=false; memMatched++;
      document.getElementById('memMatched').textContent=`${memMatched}/8`;
      if(memCards.every(c=>c.matched)){
        setTimeout(()=>{ document.getElementById('gameArea').innerHTML=`<div style="text-align:center;padding:30px;"><i class="fas fa-trophy" style="font-size:3rem;color:var(--bright-gold);"></i><h3 style="color:var(--bright-gold);">🎉 Memory Master!</h3><p style="color:var(--gold);font-size:1.2rem;font-weight:700;">Completed in ${memMoves} moves!</p><button onclick="closeGameModal()" style="margin-top:10px;padding:10px 25px;background:linear-gradient(135deg,var(--gold),var(--bright-gold));color:#000;border:none;border-radius:10px;font-weight:900;cursor:pointer;font-family:'Rajdhani',sans-serif;">Close</button></div>`; },500);
      }
    } else {
      setTimeout(()=>{
        cells[i1].textContent='❓'; cells[i1].classList.add('hidden'); cells[i1].classList.remove('flipped');
        cells[i2].textContent='❓'; cells[i2].classList.add('hidden'); cells[i2].classList.remove('flipped');
        memCards[i1].flipped=false; memCards[i2].flipped=false; memFlip=[]; memLocked=false;
      },800);
    }
  }
}

  const opts=[ans];
  while(opts.length<4){ const off=Math.floor(Math.random()*10)+1; const c=Math.random()>0.5?ans+off:ans-off; if(c>=0&&!opts.includes(c)) opts.push(c); }

function renderPuzzle(area){
  spTiles=[1,2,3,4,5,6,7,8,0]; spMoves=0;
  for(let i=0;i<150;i++){ const bi=spTiles.indexOf(0); const adj=puzzleAdj(bi); const pick=adj[Math.floor(Math.random()*adj.length)]; [spTiles[bi],spTiles[pick]]=[spTiles[pick],spTiles[bi]]; }
  area.innerHTML=`

function puzzleAdj(i){ const r=Math.floor(i/3),c=i%3,out=[]; if(r>0) out.push(i-3); if(r<2) out.push(i+3); if(c>0) out.push(i-1); if(c<2) out.push(i+1); return out; }
function drawPuzzle(){
  const grid=document.getElementById('puzzleGrid'); if(!grid) return;
  grid.innerHTML='';
  spTiles.forEach((v,i)=>{
    const cell=document.createElement('div'); cell.className='escape-cell';
    cell.style.fontSize='1.5rem'; cell.style.fontWeight='900'; cell.style.fontFamily="'Orbitron',monospace";
    if(v===0){ cell.textContent=''; cell.style.background='rgba(0,0,0,0.5)'; cell.style.borderStyle='dashed'; cell.style.cursor='default'; }
    else { cell.textContent=v; cell.style.color='var(--bright-gold)'; cell.onclick=()=>movePuzzle(i); }
    grid.appendChild(cell);
  });
  if(spTiles.every((v,i)=>v===(i+1)%9)){
    grid.innerHTML=`<div style="grid-column:1/-1;text-align:center;padding:30px;"><div style="font-size:2.8rem;">🏆</div><h3 style="color:var(--bright-gold);font-family:'Orbitron';">Solved!</h3><p style="color:var(--gold);font-weight:700;">Moves: ${spMoves}</p><button onclick="closeGameModal()" style="margin-top:10px;padding:10px 25px;background:linear-gradient(135deg,var(--gold),var(--bright-gold));color:#000;border:none;border-radius:10px;font-weight:900;cursor:pointer;font-family:'Rajdhani',sans-serif;">Close</button></div>`;
  }
}
function movePuzzle(i){
  const bi=spTiles.indexOf(0); if(!puzzleAdj(bi).includes(i)) return;
  [spTiles[bi],spTiles[i]]=[spTiles[i],spTiles[bi]]; spMoves++;

}

  globalGameKeydown=(e)=>{
    if(!document.getElementById('matchArea')) return;
    let dx=0,dy=0;
    if(e.key==='ArrowUp') dy=-2; if(e.key==='ArrowDown') dy=2;
    if(e.key==='ArrowLeft') dx=-2; if(e.key==='ArrowRight') dx=2;
    if(dx!==0||dy!==0){
      e.preventDefault();
      px=Math.max(5,Math.min(95,px+dx)); py=Math.max(5,Math.min(95,py-dy));
      const dist=Math.hypot(px-bx,py-by);
      if(dist<10){ bx+=dx*1.5; by-=dy*1.5; bx=Math.max(2,Math.min(98,bx)); by=Math.max(2,Math.min(98,by)); }
      updatePos();
    }
  };
  document.addEventListener('keydown',globalGameKeydown);
  function updatePos(){
    if(!pEl) return;
    pEl.style.bottom=py+'%'; pEl.style.left=px+'%';
    bEl.style.bottom=by+'%'; bEl.style.left=bx+'%';
    aEl.style.top=ay+'%'; aEl.style.left=ax+'%';
    if(ay>20){ if(ax<bx) ax+=0.5; if(ax>bx) ax-=0.5; if(ay<by) ay+=0.5; if(ay>by) ay-=0.5; }
    if(by>90&&bx>35&&bx<65){
      document.getElementById('rmRes').textContent='⚽ GOOOAL! You scored!';
      document.getElementById('rmRes').style.color='var(--bright-gold)';
      document.getElementById('rmRes').style.fontSize='1.6rem';
      globalGameKeydown=null; document.removeEventListener('keydown',globalGameKeydown);
    }
  }
}

function handleFiles(fileList){
  const files=Array.from(fileList); const valid=[]; const errs=[]; const MAX=5*1024*1024;
  files.forEach(f=>{ if(!f.type.startsWith('image/')){ errs.push(f.name+': not an image'); return; } if(f.size>MAX){ errs.push(f.name+': exceeds 5MB'); return; } valid.push(f); });
  const msg=document.getElementById('uploadMessage'); if(errs.length) msg.textContent='⚠️ '+errs.join(' · '); else msg.textContent='';
  valid.forEach(file=>{ const reader=new FileReader(); reader.onload=()=>{ pendingUploads.push({ file, dataUrl:reader.result, title:file.name.replace(/\.[^.]+$/,'') }); renderUploadPreview(); }; reader.readAsDataURL(file); });
}
function renderUploadPreview(){
  const grid=document.getElementById('uploadPreviewGrid');
  grid.innerHTML=pendingUploads.map((p,i)=>`<div class="upload-preview-item"><img src="${p.dataUrl}" alt=""><button class="up-remove" onclick="removeUploadPreview(${i})"><i class="fas fa-times"></i></button></div>`).join('');
  const saveBtn=document.getElementById('uploadSaveBtn'); saveBtn.disabled=pendingUploads.length===0; saveBtn.textContent=pendingUploads.length>1?`Save ${pendingUploads.length} Photos`:'Save Photo';
}
function removeUploadPreview(i){ pendingUploads.splice(i,1); renderUploadPreview(); }
async function saveUploadedPhotos(){
  if(!isAdmin()) return; if(pendingUploads.length===0) return; if(!galleryDB){ await initGalleryStorage(); }
  const title=document.getElementById('uploadTitle').value.trim(); const desc=document.getElementById('uploadDesc').value.trim(); const category=document.getElementById('uploadCategory').value||'other';
  const progress=document.getElementById('uploadProgress'); const bar=document.getElementById('uploadProgressBar'); const msg=document.getElementById('uploadMessage');
  progress.classList.add('active'); let saved=0;
  for(let i=0;i<pendingUploads.length;i++){
    const p=pendingUploads[i];
    try{ const photo={ title:title||p.title||'Untitled', description:desc, category, dataUrl:p.dataUrl, createdAt:Date.now() }; await addPhotoToDB(photo); saved++; const pct=Math.round(((i+1)/pendingUploads.length)*100); bar.style.width=pct+'%'; await new Promise(r=>setTimeout(r,80)); }
    catch(e){ msg.textContent='❌ Error saving photo: '+e.message; msg.style.color='#e05555'; progress.classList.remove('active'); return; }
  }
  msg.textContent=`✅ ${saved} photo${saved>1?'s':''} saved!`; msg.style.color='var(--bright-gold)';
  await loadPhotosFromDB(); renderGallery(); pendingUploads=[]; renderUploadPreview();
  setTimeout(()=>{ closeUploadModal(); },900);
}

function switchView(view){
  const views=['home-view','goats-view','compare-view','quiz-view','games-view','profile-view','gallery-view','learn-view','contact-view'];
  views.forEach(v=>{ const el=document.getElementById(v); if(el){ el.style.display='none'; el.className='view-hidden'; } });
  const target=document.getElementById(view+'-view');
  if(target){ if(view==='home'){ target.style.display='flex'; target.className='hero'; } else { target.style.display='block'; target.className='view-visible'; } }
  document.querySelectorAll('.nav-links a').forEach(a=>{ a.classList.remove('active'); const t=a.textContent.trim().toLowerCase(); if(t===view||(view==='goats'&&t==='players')||(view==='compare'&&t==='players')) a.classList.add('active'); });
  if(view==='profile') renderProfile();
  if(view==='goats'){ resetPlayerSearch(); }
  if(view==='compare') renderCompare();
  if(view==='quiz') renderQuizHub();
  if(view==='gallery'){ applyAdminVisibility(); renderGalleryFilters(); renderGallery(); if(isAdmin()) renderAdminGalleryList(); }
  window.scrollTo({top:0,behavior:'smooth'});
}
